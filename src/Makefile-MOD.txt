# Accumulated lists (built up by foreach/eval calls)
PP_JSI_INPUTS  :=
PP_JSI_OUTPUTS :=

# Step 1: Accumulator macro — just registers each file, defines its sed target
define PP_JSI_MULTI
PP_JSI_INPUTS  += $(1).in
PP_JSI_OUTPUTS += $(1).tmp

$(1): PP_JSI_RUN
	@cat $(1).tmp \
	    | sed -r '/\/\*{5} PCPP_END \*{5}\//{n;N;d}' \
	    | sed -r '/\/\*{5} PCPP_BEG \*{5}\//,/\/\*{5} PCPP_END \*{5}\//d' \
	    | sed 's/__NL__/\n   /g' > $(1)
endef

# Step 2: Register all files
$(foreach I,$(JAVA_SOURCES_PP),$(eval $(call PP_JSI_MULTI,$(I))))

# Step 3: The single batched pcpp-java target
#
# For each .java / .java.in pair, include in the batch only if:
#   - the .java does not exist yet, OR
#   - the .in is newer than the .java
#
# $(call NEEDS_PP, file.java) → file.java.in  if rebuild needed, else empty
NEEDS_PP = $(if $(wildcard $(1)),$(if $(filter-out $(1),$(shell find $(1).in -newer $(1) 2>/dev/null)),,$(1).in),$(1).in)

PP_JSI_RUN:
	$(eval _DIRTY_IN  := $(strip $(foreach O,$(patsubst %.tmp,%.java,$(PP_JSI_OUTPUTS)),$(call NEEDS_PP,$(O)))))
	$(eval _DIRTY_OUT := $(patsubst %.java.in,%.tmp,$(_DIRTY_IN)))
	@if [ -n "$(_DIRTY_IN)" ]; then \
	    echo "[pcpp] Processing $(words $(_DIRTY_IN)) file(s): $(_DIRTY_IN)"; \
	    java -jar pcpp-java.jar --line-directive '' --passthru-comments $(_DIRTY_IN) -o $(_DIRTY_OUT); \
	else \
	    echo "[pcpp] All files up to date, skipping pcpp-java"; \
	fi

.PHONY: PP_JSI_RUN


# USAGE #

# Replace this:
$(foreach I,$(JAVA_SOURCES_PP),$(eval $(call PP_JSI,$(I))))

# With this:
$(foreach I,$(JAVA_SOURCES_PP),$(eval $(call PP_JSI_MULTI,$(I))))
