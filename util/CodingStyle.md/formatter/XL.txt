#!/bin/bash

export NODE_PATH=/opt/node-v24.14.0-linux-x64/lib/node_modules
export PATH=/opt/node-v24.14.0-linux-x64/bin:$PATH

npm config set prefix ~/mynpm

export NODE_OPTIONS="--max-old-space-size=2048"

CG=/sys/fs/cgroup/claude_limit
if [ ! -d "$CG" ]; then
    sudo mkdir "$CG"
    sudo bash -c "echo '+memory' > /sys/fs/cgroup/cgroup.subtree_control"
    sudo bash -c "echo 4G > $CG/memory.max"
    sudo chown aloysius:aloysius "$CG" "$CG/cgroup.procs" "$CG/memory.max"
fi

trap 'kill -- -$$ 2>/dev/null; wait' EXIT

echo $$ > "$CG/cgroup.procs"
exec  ~/mynpm/bin/claude "$@"
